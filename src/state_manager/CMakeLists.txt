cmake_minimum_required(VERSION 3.0.2)
project(state_manager)
find_package(catkin REQUIRED COMPONENTS mando_localization selector planning_interfaces rospy nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs tf2_ros visualization_msgs)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS mando_localization selector planning_interfaces rospy nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs tf2_ros visualization_msgs)
catkin_install_python(PROGRAMS scripts/state_manager_node scripts/landmark_editor_node scripts/mission_inspection_node DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch rviz examples DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  catkin_install_python(PROGRAMS test/mission_pipeline_ros.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  add_rostest(test/mission_pipeline.test)
  find_package(PythonInterp 3 REQUIRED)
  add_test(NAME state_manager_unit COMMAND ${PYTHON_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test)
  set_tests_properties(state_manager_unit PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/src:${CMAKE_CURRENT_SOURCE_DIR}/../selector/src")
endif()
